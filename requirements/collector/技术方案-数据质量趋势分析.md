### 2.2 ECharts 折线图配置

每个数据类别一个折线图，4 条线（total/completeness/freshness/anomaly）：

```javascript
function renderTrendChart(containerId, data) {
    const chart = echarts.init(document.getElementById(containerId));
    chart.setOption({
        tooltip: { trigger: 'axis' },
        legend: { data: ['总分', '完整度', '新鲜度', '异常检测'] },
        xAxis: { type: 'category', data: data.map(d => d.check_time.slice(0,10)) },
        yAxis: { type: 'value', min: 0, max: 100 },
        series: [
            { name: '总分', type: 'line', data: data.map(d => d.total_score), 
              lineStyle: { width: 2.5 }, color: '#1890ff' },
            { name: '完整度', type: 'line', data: data.map(d => d.completeness),
              color: '#52c41a' },
            { name: '新鲜度', type: 'line', data: data.map(d => d.freshness),
              color: '#faad14' },
            { name: '异常检测', type: 'line', data: data.map(d => d.anomaly),
              color: '#ff4d4f' },
        ],
        grid: { left: 50, right: 20, top: 30, bottom: 30 },
    });
}
```
