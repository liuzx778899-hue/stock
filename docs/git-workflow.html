Deploy 流程（唯一 merge main 的角色）

Step 1: 部署前检查
git fetch origin && git tag | Select-String "round-.*-itest$"
git rev-parse HEAD              # 回滚点

Step 2: 拉取代码 + 合并新功能（先 merge 本地，验证后再 push）
git checkout main && git pull origin main
git merge feature/{N}-{task} --no-ff

Step 3: 停止旧服务
$procId = (Get-NetTCPConnection -LocalPort 8000).OwningProcess
if ($procId) { Stop-Process -Id $procId -Force }

Step 4: 启动新服务（跑的是 merge 后的新代码）
Start-Process -WindowStyle Hidden -FilePath python -ArgumentList web_app.py

Step 5: 部署验证（验证新代码）
curl -s http://localhost:8000/ + /api/status/stocks/quality/report/datasource/providers
# 验证失败 → git reset --hard 回滚点

Step 6: 推送上线
git tag deploy-{N} && git push origin main --tags

常用命令

# 流水线状态
git tag --list "round-*-dev"       # 待审查
git tag --list "round-*-review"    # 待集成测试
git tag --list "round-*-itest"     # 待部署

# Issues
gh issue list --label bug --state open           # 待修复 Bug
gh issue list --label enhancement --state open   # 待开发需求
gh issue list --label question --state open      # 待讨论

# 分支
git branch -r | grep -E "(feature/|fix/)"   # 活跃分支
